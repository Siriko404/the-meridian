def fine_cheapest(state, player_id):
    """Timeout fine (guide:175): burn the player's cheapest item by point value,
    ties broken by state.rng. Cap-free (a penalty, like sacrifice). None if broke."""
    p = player_by_id(state, player_id)
    if not p.inventory:
        return None
    low = min(POINT_VALUE[it] for it in p.inventory)
    ties = sorted((it for it in p.inventory if POINT_VALUE[it] == low), key=lambda it: it.name)
    item = state.rng.choice(ties)
    p.inventory[item] -= 1
    if p.inventory[item] == 0:
        del p.inventory[item]
    return item
```

def _post_swap(inv, gives, gets):
    out = dict(inv)
    for it in gives:
        if out.get(it, 0) <= 0:
            raise ValueError("cannot give unowned item")
        out[it] -= 1
        if out[it] == 0:
            del out[it]
    for it in gets:
        out[it] = out.get(it, 0) + 1
    if any(c > MAX_PER_TYPE for c in out.values()) or sum(out.values()) > MAX_TOTAL:
        raise ValueError("swap breaks inventory caps")
    return out


def swap(state, a_id, b_id, give_a, give_b):
    """All-or-nothing two-way transfer (spec 4.5, audit a5): validated against the
    POST-swap inventories; no mid-flight can_receive re-checks; nothing moves on failure."""
    a = player_by_id(state, a_id)
    b = player_by_id(state, b_id)
    new_a = _post_swap(a.inventory, give_a, give_b)
    new_b = _post_swap(b.inventory, give_b, give_a)
    a.inventory = new_a
    b.inventory = new_b
```

    def __init__(self, human_seats, seed, num_players=5, now_fn=time.time):

    def _grace(self):

                       for e in self.last_events), 20.0)

    def _begin_phase_for_current(self):
        self.gen += 1
        if self.winner is not None:
            self.phase = None
            self.phase_ends_at = None
            return
        dur = self.DECISION_S if self.state.bonus_pending else self.TALK_S
        self.phase = "decision" if self.state.bonus_pending else "talk"

            raise IllegalAction("not in talk phase")
        self.gen += 1
        self.phase = "decision"
        self.phase_ends_at = self.now() + self.DECISION_S
        self.last_events = []

    def trade_offer(self, seat, to, give, want):
        if self.winner is not None or seat != self.state.current_id:
            raise IllegalAction("not your turn")
        if self.phase != "talk":
            raise IllegalAction("trades happen in the talk phase")
        if self.pending_offer is not None:
            raise IllegalAction("an offer is already pending")
        me = player_by_id(self.state, seat)
        mate = player_by_id(self.state, to)
        if not mate.alive or to == seat or mate.team != me.team:
            raise IllegalAction("can only trade with a living teammate")
        for n in list(give) + list(want):
            if n not in Item.__members__:
                raise IllegalAction("bad item name")
        self.gen += 1
        self.pending_offer = {"from": seat, "to": to, "give": list(give), "want": list(want)}

    def trade_withdraw(self, seat):
        if self.pending_offer is None or self.pending_offer["from"] != seat:
            raise IllegalAction("no offer of yours is pending")
        self.gen += 1
        self.pending_offer = None

    def trade_respond(self, seat, accept):
        if self.pending_offer is None or self.pending_offer["to"] != seat:
            raise IllegalAction("no offer for you is pending")
        off = self.pending_offer
        self.gen += 1
        self.pending_offer = None
        if not accept:
            self._trade_result = {"kind": "declined", "parties": (off["from"], off["to"])}
            return
        give = [Item[n] for n in off["give"]]
        want = [Item[n] for n in off["want"]]
        try:

        except ValueError:

            return

        self._trade_result = {"kind": "settled", "parties": (off["from"], off["to"])}

async def _deadline(lb, gen):
    try:
        await asyncio.sleep(max(0.0, lb.room.phase_ends_at - time.time()))
    except asyncio.CancelledError:
        return
    async with lb.lock:
        if lb.room is None or lb.room.gen != gen:

        acted = lb.room.on_deadline()
    if acted:
        await _broadcast_views(lb)

def _schedule_deadline(lb):
    if lb.deadline_task is not None:

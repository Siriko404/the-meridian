## Landed protocol (primary-source truth — `room.py` / `view.py` / `server.py` on master)

**Client→server:** `join {code, name, token}` · `start` · `action {i}` · `proceed` · `trade_offer {to, give:[ENUM], want:[ENUM]}` · `trade_respond {accept}` · `trade_withdraw` · `flip {target, items:[ENUM]}` · `chat {text}` · `ping`.

**Event payloads (public; per-seat enrichment noted):**
`announce {total, live}` · `reload {total, live}` · `shot {actor, target, shell:"LIVE"|"BLANK", damage, killed}` · `elimination {actor:killer, target:victim}` · `item {actor, item:ENUM, target?}` · `flip {actor, target}` · `fine {actor}` (no item, D14) · `draw {you?:{attempted:[ENUM],received:[ENUM]}}` (public copy empty) · `skip {target}` · `trade {actor, target, items?:{give,want}}` (items = parties only) · `trade_private {kind:"declined"|"cancelled"}` (parties only) · `gift {actor, target}` (bot path, no items — constraint 11).

```js
/* Dice-ceremony synthesis (spec M2/M16): fabricate hi/lo d6 rolls consistent with
 * the ANNOUNCED totals — the real rolls never reach the wire (spec §4.6, audit a3).
 * Pure + node-tested. Feasibility: k=ceil(total/6) dice; live>=0.3*total => k<=live. */
(function (global) {
  "use strict";
  function spread(sum, k, cap) {            // k values in 1..cap summing to `sum`, near-even
    const out = Array(k).fill(1);
    let rest = sum - k, guard = 6 * k;
    for (let i = 0; rest > 0 && guard > 0; i = (i + 1) % k, guard--) {
      if (out[i] < cap) { out[i]++; rest--; }
    }
    return out;
  }
  function synthDice(total, live) {
    total = Math.max(2, total | 0);
    const k = Math.ceil(total / 6);
    live = Math.min(Math.max(k, live | 0), total);   // clamp: k<=live<=total keeps it solvable
    const his = spread(total, k, 6);
    const los = Array(k).fill(1);
    let rest = live - k, guard = 6 * k;
    for (let i = 0; rest > 0 && guard > 0; i = (i + 1) % k, guard--) {
      if (los[i] < his[i]) { los[i]++; rest--; }     // per-slot cap = its hi => lo<=hi always
    }
    return his.map((h, i) => [h, los[i]]);
  }
  const DICE = { synthDice };
  global.DICE = DICE;
  if (typeof module !== "undefined" && module.exports) module.exports = DICE;
})(typeof window !== "undefined" ? window : globalThis);
```

```js
/* WS glue + the gen gate + the countdown clock. Pure parts node-tested.
 * gen gate: the view carries `gen`; events are REPLAYED only when gen increased —
 * a join/reconnect rebroadcast must settle silently, a stale view must never rewind. */
(function (global) {
  "use strict";
  function genGate() {
    let last = -1;
    return function accept(view) {
      const g = typeof view.gen === "number" ? view.gen : -1;
      if (g > last) { last = g; return true; }
      return false;
    };
  }
  function countdown(view, nowMs) {
    // anchor the server↔client clock offset ONCE per view; recompute absolute remaining
    // every tick from wall clock (rAF throttles in background tabs — constraint 17).
    const off = (view.server_now || 0) - (nowMs == null ? Date.now() : nowMs) / 1000;
    return function left(tMs) {
      if (view.phase_ends_at == null) return null;
      const now = (tMs == null ? Date.now() : tMs) / 1000 + off;
      return Math.max(0, view.phase_ends_at - now);
    };
  }
  function connect(makeWS, onMsg) {
    const ws = makeWS();
    const send = (m) => { if (ws.readyState === 1) ws.send(JSON.stringify(m)); };
    ws.onmessage = (ev) => onMsg(JSON.parse(ev.data));
    return { ws, send };
  }
  const NET = { genGate, countdown, connect };
  global.NET = NET;
  if (typeof module !== "undefined" && module.exports) module.exports = NET;
})(typeof window !== "undefined" ? window : globalThis);
```

```js
/* CHOREO — the moment catalog (spec §7). Pure sequencing over scene primitives.
 * Skippable: CHOREO.skip() flips a flag; every wait routes through it (v1 behavior). */
(function (global) {
  "use strict";
  let _skip = false;
  function bind(scene) {
    _skip = false;
    scene.w = (ms) => new Promise((r) => {
      if (_skip || ms <= 0) { r(); return; }
      scene.time.delayedCall(ms, r);
    });
  }
  const nm = (id, roster) => THEME.seatName(id, roster);
  const iname = (en) => THEME.itemName(en);

  /* ---------- setup ceremony: M1 coin, plates stamp, M2+M3 load, M4 deal, M5 first turn ---------- */
  async function setup(scene, view, roster, announce) {
    bind(scene);
    scene.render(view, roster);
    view.order.forEach((id) => { const s = scene.seats.get(id);
      [s.plate, s.nameT].forEach((o) => o.setScale(0)); });
    // M1 — coin flip lands on the extra-seat agency (odd split is derivable from the view)
    const counts = {};
    view.players.forEach((p) => { counts[p.team] = (counts[p.team] || 0) + 1; });
    const odd = Object.keys(counts).sort((a, b) => counts[b] - counts[a])[0];
    await coinFlip(scene, odd);
    for (const id of view.order) {                            // plates stamp in, 120ms stagger
      const s = scene.seats.get(id);
      scene.tweens.add({ targets: [s.plate, s.nameT], scaleX: 1, scaleY: 1, duration: 160, ease: "Back.easeOut" });
      await scene.w(120);
    }
    if (announce) {                                           // M2 dice ceremony + M3 announce
      await scene.diceCeremony(announce.total, announce.live, {});
      await scene.announceHold(announce.total, announce.live);
    }
    await deal(scene, view);                                  // M4
    await scene.dialTo(view.current, false);                  // M5 first heavy click
    scene.render(view, roster);
  }

  async function coinFlip(scene, oddTeam) {
    const tm = THEME.TEAMS[oddTeam] || THEME.TEAMS.RED;
    const c = scene.add.circle(SCENE.W / 2, 210, 24, 0xffffff).setStrokeStyle(3, 0xffffff, 1).setDepth(18);
    for (let k = 0; k < 6; k++) {
      await new Promise((r) => scene.tweens.add({ targets: c, scaleX: 0.06, duration: 90, yoyo: true,
        onYoyo: () => c.setFillStyle(k % 2 ? 0xffffff : 0x000000), onComplete: r }));
    }
    c.setFillStyle(tm.plate);
    const t = scene.add.text(SCENE.W / 2, 252, tm.name + " +1 SEAT", { fontFamily: "Consolas,monospace",
      fontSize: "14px", color: "#ffffff", fontStyle: "bold" }).setOrigin(0.5).setResolution(2).setDepth(18);
    await scene.w(1100);
    scene.tweens.add({ targets: [c, t], alpha: 0, duration: 350,
      onComplete: () => { c.destroy(); t.destroy(); } });
  }

  async function quickTumble(scene, x, y) {                   // M4's per-seat 3 quick d6 tumbles
    const t = scene.add.text(x, y - 52, "", { fontFamily: "Consolas,monospace", fontSize: "16px",
      color: "#ffffff", fontStyle: "bold" }).setOrigin(0.5).setResolution(2).setDepth(15);
    for (let k = 0; k < 3; k++) { t.setText("⚀⚁⚂⚃⚄⚅"[(Math.random() * 6) | 0]); await scene.w(90); }
    t.destroy();
  }

  async function deal(scene, view) {
    const c = { x: SCENE.CX, y: SCENE.CY };
    for (const id of view.order) {
      const p = view.players.find((q) => q.id === id);
      if (!p || !p.alive) continue;
      await quickTumble(scene, scene.seatXY(id).x, scene.seatXY(id).y);
      if (id === view.you) {                                  // your icons fly into KIT, names revealed
        for (const [en, cnt] of Object.entries(view.your_hand || {}))
          for (let k = 0; k < cnt; k++)
            await scene.iconFly(THEME.ITEMS[en].icon, c, scene.kitXY(), iname(en));
      } else {
        for (let k = 0; k < 3; k++) scene.cardBack(c, id);    // 3 card-backs fan, no await between
        await scene.w(500);
      }
    }
  }

  /* ---------- the live cascade ---------- */
  async function play(scene, events, prev, view, roster) {
    bind(scene);
    let sawElim = false, lastBeer = null;
    for (const e of events || []) {
      if (e.type === "announce") {
        await scene.diceCeremony(e.total, e.live, {});
        await scene.announceHold(e.total, e.live);
      } else if (e.type === "reload") {
        await scene.diceCeremony(e.total, e.live, { fast: true, honor: sawElim });
        await scene.announceHold(e.total, e.live);
        if (lastBeer != null && lastBeer === view.current && view.winner == null)
          await scene.toast("STILL YOUR SHOT", 1400);         // C3 after a MAGNET-emptied reload
      } else if (e.type === "shot") {
        await scene.dialTo(e.actor, true);                    // M12 overwind + lock
        await scene.w(400);                                   // tension hold
        if (e.shell === "LIVE") {
          scene.flashWhite(); scene.shake();
          scene.seatRecoil(e.target);
          scene.pipShatter(e.target, e.damage || 1);
          scene.floatText(e.target, "−" + (e.damage || 1), true);
          await scene.w(1000);
        } else {
          scene.tweens.add({ targets: scene.center, angle: "+=3", yoyo: true, repeat: 3, duration: 40 });
          scene.floatText(e.actor, "CLICK", false);
          await scene.w(800);
          if (e.actor === e.target && view.bonus && view.current === e.actor && view.winner == null)
            await scene.stampSeat(e.actor, "BONUS TURN · SILENT", false);        // M13
        }
        scene.setSawChip(false); lastBeer = null;
      } else if (e.type === "elimination") {
```

```js
      const it = queue.shift();
      const prev = curView;
      curView = it.view;
      leftFn = NET.countdown(curView);

        await CHOREO.setup(scene, curView, roster, ann);       // M1–M5

        await CHOREO.play(scene, it.events, prev, curView, roster);
      } else {
        scene.render(curView, roster);                         // stale/rebroadcast: settle only

      noteOverride = null;
      afterSettle();                                           // Task 7: rebuild verbs/trays
      tick();

  function computeNote(v) {
```

```js
(function (global) {
  "use strict";
  const MONO = "Consolas,monospace";
  const st = (size, color) => ({ fontFamily: MONO, fontSize: size + "px",
    color: color || "#ffffff", fontStyle: "bold", align: "center" });
  const ORDER = ["BEER", "HANDCUFFS", "MAGNIFIER", "SAW", "CIGARETTES", "IMMUNITY"];
  let open = null;

  function close() { if (open) { open.cont.destroy(); open = null; } }
  function tray(scene, name, h) {
    close();
    const y = 844 - h / 2 - 4;
    const cont = scene.add.container(0, 0).setDepth(22);
    cont.add(scene.add.rectangle(SCENE.W / 2, y, SCENE.W - 8, h, 0x000000, 0.97)
      .setStrokeStyle(2, 0xffffff, 0.95).setInteractive());     // swallows taps under the sheet
```

```js
  /* ---- [ACT] interaction grammar (§6.3): verb → picker → red brackets → confirm ---- */
  function myTurn() {
    return curView && curView.winner == null && curView.current === curView.you && !busy;
  }
  function meAlive() {
    const p = curView && curView.players.find((q) => q.id === curView.you);
    return !!(p && p.alive);
  }
  function teammates() {
    if (!curView.teams_active) return [];
    const me = curView.players.find((p) => p.id === curView.you);
    return curView.players.filter((p) => p.alive && p.id !== curView.you && p.team === me.team)
      .map((p) => p.id);
  }

  function afterSettle() {
    noteOverride = null;
    scene.clearTargets(); scene.seatGlow([], false); TRAYS.close();
    scene.onKit = () => TRAYS.kit(scene, curView, usableOpts());     // info mode off-turn
    scene.onLedger = () => TRAYS.ledger(scene, curView);
    scene.onPhone = () => $("chatsheet").classList.toggle("on");
    const chatOk = curView.phase === "talk" && meAlive() && curView.winner == null;
    $("chatin").disabled = !chatOk;
    $("chatin").placeholder = chatOk ? "say something…" : "SILENCE.";
    buildStrip();
    const po = curView.pending_offer;                                // incoming offer → tray now
    if (po && po.to === curView.you) {                               // (rides the view: reconnect-safe)
      TRAYS.respond(scene, curView, roster,
        () => { TRAYS.close(); net.send({ type: "trade_respond", accept: true }); },
        () => { TRAYS.close(); net.send({ type: "trade_respond", accept: false }); });
    }
  }

  function usableOpts() {
    if (!myTurn()) return {};
    const usable = new Set((curView.legal || []).filter((a) => a.kind === "item").map((a) => a.item));
    return { usable, onPick: pickItemTargets };
  }

  function buildStrip() {
    if (!myTurn()) { scene.setStrip([]); return; }
    const kinds = new Set((curView.legal || []).map((a) => a.kind));
    const po = curView.pending_offer;
    const defs = [];
    if (curView.phase === "talk") {
      if (kinds.has("item")) defs.push({ label: "ITEM", onTap: () => TRAYS.kit(scene, curView, usableOpts()) });
      if (po && po.from === curView.you)
        defs.push({ label: "WITHDRAW", onTap: () => net.send({ type: "trade_withdraw" }), wide: true });
      else if (teammates().length) defs.push({ label: "TRADE", onTap: pickTrade });
      defs.push({ label: "PROCEED", onTap: () => { wait(); net.send({ type: "proceed" }); }, wide: true });
    } else if (curView.phase === "decision") {
      if (kinds.has("shoot")) defs.push({ label: "SHOOT", onTap: () => pickTarget("shoot") });
      if (kinds.has("flip")) defs.push({ label: "FLIP", onTap: () => pickTarget("flip") });
      if (kinds.has("item")) defs.push({ label: "ITEM", onTap: () => TRAYS.kit(scene, curView, usableOpts()) });
    }
    scene.setStrip(defs);
  }

  function cancelSel() {
    noteOverride = null;
    scene.clearTargets(); scene.seatGlow([], false); TRAYS.close();
    buildStrip(); tick();
  }
  function confirmAct(label, go) {
    scene.setStrip([{ label: label + " — CONFIRM", onTap: go, wide: true },
                    { label: "CANCEL", onTap: cancelSel }]);
  }
  function wait() { scene.clearTargets(); scene.seatGlow([], false); scene.setStrip([]); }
  function sendAction(i) { wait(); TRAYS.close(); net.send({ type: "action", i }); }

  function pickItemTargets(en) {
    const entries = curView.legal.filter((a) => a.kind === "item" && a.item === en);
    const nt = entries.find((a) => a.target == null);
    if (nt) { confirmAct("USE " + THEME.itemName(en), () => sendAction(nt.i)); return; }
    scene.setTargets(entries.map((a) => a.target), (id) => {
      const a = entries.find((x) => x.target === id);
      scene.clearTargets();
      confirmAct(THEME.itemName(en) + " → " + THEME.seatName(id, roster), () => sendAction(a.i));
    });
  }

  function pickTarget(kind) {
    const entries = curView.legal.filter((a) => a.kind === kind);
    const ids = entries.map((a) => a.target);
    if (kind === "shoot" && ids.length === 1 && ids[0] === curView.you) {     // M11/L4
      noteOverride = "NO LEGAL TARGETS — YOU FACE THE GUN (FLIP SOMEONE TO CHANGE THAT)";
      tick();
    }
    scene.setTargets(ids, (id) => {
      const a = entries.find((x) => x.target === id);
      scene.clearTargets();
      if (kind === "flip") {
        TRAYS.pay(scene, curView, id, a.cost, roster,
          (items) => { TRAYS.close(); wait(); net.send({ type: "flip", target: id, items }); },
          cancelSel);
      } else {
        const tgt = id === curView.you ? "YOURSELF" : THEME.seatName(id, roster);
        confirmAct("SHOOT " + tgt, () => sendAction(a.i));
      }
    });
  }

  function pickTrade() {                                          // M26 — glow, NOT red brackets
    scene.seatGlow(teammates(), true, (id) => {
      scene.seatGlow([], false);
      TRAYS.offer(scene, curView, id, roster,
        (give, want) => { TRAYS.close(); net.send({ type: "trade_offer", to: id, give, want }); buildStrip(); },
        cancelSel);
    });
  }
```
